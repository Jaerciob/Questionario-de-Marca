        function showQuestion() {
            const questionData = getQuestionData(currentQuestionIndex);
            if (!questionData) return;

            // Update progress
            const progress = ((currentQuestionIndex + 1) / totalQuestions) * 100;
            document.getElementById('progress-bar').style.width = progress + '%';
            document.getElementById('progress-text').textContent = `Questão ${currentQuestionIndex + 1} de ${totalQuestions}`;
            document.getElementById('progress-percentage').textContent = `${Math.round(progress)}% concluído`;

            // Update question content
            document.getElementById('section-title').textContent = questionData.section;
            document.getElementById('question-text').textContent = questionData.question.text;

            // Generate options
            const optionsContainer = document.getElementById('options-container');
            optionsContainer.innerHTML = '';

            questionData.question.options.forEach((option, index) => {
                const optionDiv = document.createElement('div');
                optionDiv.className = 'option-item bg-white p-4 rounded-lg border-2 border-gray-200 hover:border-0367A6';
                optionDiv.innerHTML = `
                    <label class="flex items-center cursor-pointer w-full">
                        <input type="radio" name="question${currentQuestionIndex}" value="${index}" class="mr-4 text-0367A6 hidden">
                        <span class="text-gray-900">${option}</span>
                    </label>
                `;

                // Add event listener to the option div
                optionDiv.addEventListener('click', function() {
                    // Remove selection from all options
                    document.querySelectorAll('.option-item').forEach(item => {
                        item.classList.remove('option-selected');
                        item.querySelector('input').checked = false;
                    });
                    // Add selection to the clicked option
                    this.classList.add('option-selected');
                    const radioInput = this.querySelector('input');
                    radioInput.checked = true;
                    // Store the answer in the global array
                    answers[currentQuestionIndex] = parseInt(radioInput.value);

                    // Show or hide the "other specialty" input field
                    if (option === 'Outro') {
                        showOtherSpecialtyInput();
                    } else {
                        hideOtherSpecialtyInput();
                    }
                });
                
                optionsContainer.appendChild(optionDiv);

                // Re-select the previously answered option
                if (answers[currentQuestionIndex] === index) {
                    optionDiv.classList.add('option-selected');
                    optionDiv.querySelector('input').checked = true;
                    if (option === 'Outro') {
                        showOtherSpecialtyInput();
                    }
                }
            });

            // If the current question is about 'other' specialty and an answer exists, display it
            if (questionData.question.text.includes("área de atuação") && answers[currentQuestionIndex] === questionData.question.options.length - 1) {
                showOtherSpecialtyInput();
            }

            // Update navigation buttons
            const prevBtn = document.getElementById('prev-btn');
            const nextBtn = document.getElementById('next-btn');

            if (currentQuestionIndex === 0) {
                prevBtn.classList.add('hidden');
            } else {
                prevBtn.classList.remove('hidden');
            }

            if (currentQuestionIndex === totalQuestions - 1) {
                nextBtn.innerHTML = '<i class="fas fa-chart-line mr-2"></i>Ver Resultados';
            } else {
                nextBtn.innerHTML = 'Próxima<i class="fas fa-arrow-right ml-2"></i>';
            }
        }

        function showOtherSpecialtyInput() {
            const optionsContainer = document.getElementById('options-container');
            let otherInput = document.getElementById('other-specialty-input');
            if (!otherInput) {
                otherInput = document.createElement('input');
                otherInput.type = 'text';
                otherInput.id = 'other-specialty-input';
                otherInput.placeholder = 'Digite a área de atuação';
                otherInput.className = 'mt-4 bg-white bg-opacity-10 border-gray-200 border-2 rounded-lg p-3 w-full text-gray-900';
                otherInput.addEventListener('input', function() {
                    otherSpecialty = this.value;
                });
                optionsContainer.appendChild(otherInput);
            }
            if(otherSpecialty) {
                otherInput.value = otherSpecialty;
            }
            
        }

        function hideOtherSpecialtyInput() {
            const otherInput = document.getElementById('other-specialty-input');
            if (otherInput) {
                otherInput.remove();
            }
        }

        function getQuestionData(questionIndex) {
            let currentIndex = 0;
            for (let section of quizData.sections) {
                for (let question of section.questions) {
                    if (currentIndex === questionIndex) {
                        return {
                            section: section.title,
                            question: question
                        };
                    }
                    currentIndex++;
                }
            }
            return null;
        }

        function nextQuestion() {
            const selectedAnswer = document.querySelector(`input[name="question${currentQuestionIndex}"]:checked`);
            if (!selectedAnswer) {
                console.error('Por favor, selecione uma resposta antes de continuar.');
                return;
            }

            if (selectedAnswer.value === (quizData.sections[1].questions[1].options.length -1).toString() && !otherSpecialty) {
                alert('Por favor, digite a sua área de atuação em "Outro" antes de continuar.');
                return;
            }

            // The answer is already stored on the option click, just advance
            if (currentQuestionIndex < totalQuestions - 1) {
                currentQuestionIndex++;
                showQuestion();
            } else {
                showResults();
            }
        }

        function previousQuestion() {
            if (currentQuestionIndex > 0) {
                hideOtherSpecialtyInput();
                currentQuestionIndex--;
                showQuestion();
            }
        }

        function calculateResults() {
            let totalScore = 0;
            // Each question is worth 20 points, with 5 options (1 to 5).
            answers.forEach((answer) => {
                totalScore += (answer + 1) * 4;
            });

            let category, categoryColor;
            if (totalScore >= 180) {
                category = "Excelente";
                categoryColor = "text-0367A6";
            } else if (totalScore >= 140) {
                category = "Muito Bom";
                categoryColor = "text-F2A007";
            } else if (totalScore >= 100) {
                category = "Bom";
                categoryColor = "text-yellow-500";
            } else if (totalScore >= 60) {
                category = "Regular";
                categoryColor = "text-BF5B04";
            } else {
                category = "Precisa Melhorar";
                categoryColor = "text-red-600";
            }

            // Archetype selection based on specific answers
            const archetypeNames = ["Sábio", "Criador", "Governante", "Prestativo"];
            const archetypeScore = new Array(archetypeNames.length).fill(0);
            
            // Simplified scoring logic for archetypes
            archetypeScore[0] += answers[0] === 0 ? 1 : 0; // Tradition and solidity
            archetypeScore[0] += answers[1] === 0 ? 1 : 0; // Corporate law
            archetypeScore[0] += answers[2] === 1 ? 1 : 0; // Old visual identity
            archetypeScore[0] += answers[3] === 0 ? 1 : 0; // Formal tone
            archetypeScore[0] += answers[4] === 1 ? 1 : 0; // Tradition

            archetypeScore[1] += answers[0] === 1 ? 1 : 0; // Innovation and agility
            archetypeScore[1] += answers[1] === 4 ? 1 : 0; // Digital law
            archetypeScore[1] += answers[2] === 4 ? 1 : 0; // Unique identity
            archetypeScore[1] += answers[3] === 3 ? 1 : 0; // Inspiring tone
            archetypeScore[1] += answers[4] === 4 ? 1 : 0; // Technology and innovation

            archetypeScore[2] += answers[0] === 3 ? 1 : 0; // Specialization
            archetypeScore[2] += answers[1] === 0 ? 1 : 0; // Corporate law
            archetypeScore[2] += answers[2] === 3 ? 1 : 0; // Modern and consistent
            archetypeScore[2] += answers[3] === 0 ? 1 : 0; // Formal tone
            archetypeScore[2] += answers[4] === 2 ? 1 : 0; // Niche market

            archetypeScore[3] += answers[0] === 2 ? 1 : 0; // Relationship and empathy
            archetypeScore[3] += answers[1] === 3 ? 1 : 0; // Family law
            archetypeScore[3] += answers[2] === 2 ? 1 : 0; // Functional identity
            archetypeScore[3] += answers[3] === 4 ? 1 : 0; // Empathetic tone
            archetypeScore[3] += answers[4] === 3 ? 1 : 0; // Humanized service
            
            const maxScore = Math.max(...archetypeScore);
            const selectedArchetype = archetypeNames[archetypeScore.indexOf(maxScore)];
            
            // Dimension calculation (corrected for the new number of questions)
            const dimensions = {
                'Identidade de Marca': Math.round((answers.slice(0, 4).reduce((a, b) => a + b + 1, 0) / 4) * 20),
                'Posicionamento': Math.round((answers.slice(4, 8).reduce((a, b) => a + b + 1, 0) / 4) * 20),
                'Comunicação': Math.round((answers.slice(8, 10).reduce((a, b) => a + b + 1, 0) / 2) * 20),
                'Experiência': Math.round((answers.slice(10, 12).reduce((a, b) => a + b + 1, 0) / 2) * 20),
                'Cultura e Valores': Math.round((answers.slice(12, 14).reduce((a, b) => a + b + 1, 0) / 2) * 20)
            };

            return {
                totalScore,
                category,
                categoryColor,
                archetype: selectedArchetype,
                dimensions,
                firmInfo: { firmName, firmSpecialty: otherSpecialty || firmSpecialty, firmTarget, firmMission }
            };
        }

            const sortedDimensions = Object.entries(dimensions).sort(([, a], [, b]) => b - a);

            const highestScore = sortedDimensions[0];
            const lowestScore = sortedDimensions[sortedDimensions.length - 1];

            const analysisTitle = document.createElement('h3');
            analysisTitle.className = 'text-lg md:text-xl font-bold mb-4 text-gray-800';
            analysisTitle.innerHTML = 'Análise Detalhada do Gráfico de Teia';
            analysisContainer.appendChild(analysisTitle);

            const strengths = document.createElement('div');
            strengths.className = 'bg-blue-100 p-4 rounded-lg mb-4';
            // Adjusted text color for better contrast
            strengths.innerHTML = `
                <h4 class="font-semibold text-blue-900 mb-2">Pontos Fortes:</h4>
                <p class="text-blue-900">
                    Sua área de maior performance é <strong>${highestScore[0]}</strong> com uma pontuação de ${highestScore[1]}%. Isso indica que seu escritório já possui uma base sólida e bem desenvolvida nesta dimensão. Continue capitalizando sobre este ponto forte para se diferenciar no mercado.
                </p>
            `;
            analysisContainer.appendChild(strengths);

            const weaknesses = document.createElement('div');
            weaknesses.className = 'bg-yellow-100 p-4 rounded-lg';
            weaknesses.innerHTML = `
                <h4 class="font-semibold text-yellow-800 mb-2">Áreas para Melhoria:</h4>
                <p class="text-gray-900">
                    A dimensão de <strong>${lowestScore[0]}</strong>, com uma pontuação de ${lowestScore[1]}%, representa a maior oportunidade de crescimento para o seu branding. Focar nesta área pode trazer um retorno significativo e fortalecer a sua marca de forma mais completa. As recomendações abaixo foram ajustadas para priorizar esta dimensão.
                </p>
            `;
            analysisContainer.appendChild(weaknesses);
        }

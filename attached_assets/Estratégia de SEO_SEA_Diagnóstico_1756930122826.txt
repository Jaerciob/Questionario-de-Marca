// SEO Quiz Functions
function renderSEOQuestion() {
    const question = seoQuestions[currentSEOQuestion];
    const container = document.getElementById('seo-question-container');
    
    container.innerHTML = `
        <div class="mb-6">
            <h3 class="text-xl font-semibold text-gray-800 mb-4">${question.question}</h3>
            <div class="space-y-3">
                ${question.options.map(option => `
                    <label class="flex items-center p-4 bg-gray-50 rounded-lg hover:bg-green-50 cursor-pointer transition-all border-2 border-transparent hover:border-green-200">
                        <input type="radio" name="seo-q${question.id}" value="${option.value}" class="mr-3 text-green-600">
                        <span class="text-gray-700">${option.text}</span>
                    </label>
                `).join('')}
            </div>
        </div>
    `;
    
    // Pre-select previous answer if exists
    if (seoAnswers[question.id]) {
        const radio = container.querySelector(`input[value="${seoAnswers[question.id]}"]`);
        if (radio) radio.checked = true;
    }
}

function updateSEOProgress() {
    const progress = ((currentSEOQuestion + 1) / seoQuestions.length) * 100;
    document.getElementById('seo-progress-bar').style.width = progress + '%';
    document.getElementById('seo-progress-text').textContent = `${currentSEOQuestion + 1}/${seoQuestions.length}`;
    
    // Update navigation buttons
    const prevBtn = document.getElementById('seo-prev-btn');
    const nextBtn = document.getElementById('seo-next-btn');
    
    if (currentSEOQuestion === 0) {
        prevBtn.classList.add('hidden');
    } else {
        prevBtn.classList.remove('hidden');
    }
    
    if (currentSEOQuestion === seoQuestions.length - 1) {
        nextBtn.innerHTML = 'Ver Resultado <i class="fas fa-chart-line ml-2"></i>';
    } else {
        nextBtn.innerHTML = 'Próxima <i class="fas fa-arrow-right ml-2"></i>';
    }
}

function nextSEOQuestion() {
    const selectedOption = document.querySelector(`input[name="seo-q${seoQuestions[currentSEOQuestion].id}"]:checked`);
    
    if (!selectedOption) {
        alert('Por favor, selecione uma opção antes de continuar.');
        return;
    }
    
    seoAnswers[seoQuestions[currentSEOQuestion].id] = selectedOption.value;
    
    if (currentSEOQuestion < seoQuestions.length - 1) {
        currentSEOQuestion++;
        renderSEOQuestion();
        updateSEOProgress();
    } else {
        showSEOResults();
    }
}

function previousSEOQuestion() {
    if (currentSEOQuestion > 0) {
        currentSEOQuestion--;
        renderSEOQuestion();
        updateSEOProgress();
    }
}

// Results Analysis Functions
function analyzeSEOAnswers() {
    const scores = {
        maturity: 0,
        content: 0,
        technical: 0,
        authority: 0,
        conversion: 0,
        local: 0,
        budget: 0,
        time: 0
    };
    
    // Calculate scores based on answers
    seoQuestions.forEach((question, index) => {
        const answer = seoAnswers[question.id];
        const selectedOption = question.options.find(opt => opt.value === answer);
        
        if (selectedOption && selectedOption.points) {
            Object.keys(selectedOption.points).forEach(key => {
                if (scores.hasOwnProperty(key)) {
                    scores[key] += selectedOption.points[key];
                }
            });
        }
    });
    
    // Determine maturity level
    let maturityLevel = 'Iniciante';
    if (scores.maturity >= 8) maturityLevel = 'Avançado';
    else if (scores.maturity >= 4) maturityLevel = 'Emergente';
    
    // Determine strategic focus
    const focusAreas = ['content', 'technical', 'authority', 'conversion', 'local'];
    let strategicFocus = focusAreas.reduce((a, b) => scores[a] > scores[b] ? a : b);
    
    const focusMap = {
        content: 'Conteúdo',
        technical: 'Técnico',
        authority: 'Autoridade',
        conversion: 'Conversão',
        local: 'Local SEO'
    };
    
    // Estimate timeline
    let estimatedTime = '3-6 meses';
    if (scores.maturity >= 6 && scores.budget >= 2) estimatedTime = '2-4 meses';
    else if (scores.maturity <= 2) estimatedTime = '6-12 meses';
    
    return {
        maturityLevel,
        strategicFocus: focusMap[strategicFocus],
        estimatedTime,
        scores
    };
}

function analyzeSEAAnswers() {
    const scores = {
        experience: 0,
        budget: 0,
        search: 0,
        social: 0,
        display: 0,
        video: 0,
        mobile: 0,
        b2b: 0,
        tracking: 0,
        time: 0
    };
    
    // Calculate scores based on answers
    seaQuestions.forEach((question, index) => {
        const answer = seaAnswers[question.id];
        const selectedOption = question.options.find(opt => opt.value === answer);
        
        if (selectedOption && selectedOption.points) {
            Object.keys(selectedOption.points).forEach(key => {
                if (scores.hasOwnProperty(key)) {
                    scores[key] += selectedOption.points[key];
                }
            });
        }
    });
    
    // Determine experience level
    let experienceLevel = 'Iniciante';
    if (scores.experience >= 10) experienceLevel = 'Avançado';
    else if (scores.experience >= 5) experienceLevel = 'Intermediário';
    
    // Determine platform focus
    const platforms = ['search', 'social', 'display', 'video'];
    let platformFocus = platforms.reduce((a, b) => scores[a] > scores[b] ? a : b);
    
    const platformMap = {
        search: 'Google Ads',
        social: 'Social Media',
        display: 'Display Ads',
        video: 'YouTube Ads'
    };
    
    // Budget range
    let budgetRange = 'Até R$ 500';
    if (scores.budget >= 10) budgetRange = 'R$ 10.000+';
    else if (scores.budget >= 6) budgetRange = 'R$ 2.000-10.000';
    else if (scores.budget >= 4) budgetRange = 'R$ 500-2.000';
    
    return {
        experienceLevel,
        platformFocus: platformMap[platformFocus],
        budgetRange,
        scores
    };
}
